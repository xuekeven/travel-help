### Step 5：实现日期工具函数

创建 `lib/date.ts`：

```typescript
import { format, differenceInCalendarDays, addDays } from 'date-fns'
import { toZonedTime } from 'date-fns-tz'
import { zhCN } from 'date-fns/locale'

export const TIMEZONE = 'Asia/Shanghai'

// 获取当前中国时区的日期字符串，格式 'YYYY-MM-DD'
export function getTodayCN(): string {
  const zonedNow = toZonedTime(new Date(), TIMEZONE)
  return format(zonedNow, 'yyyy-MM-dd')
}

// 根据出行日期计算开售日（出行日期 - 14 天）
export function getSaleDate(departureDate: Date | string): string {
  const d = typeof departureDate === 'string' ? new Date(departureDate) : departureDate
  // 注意：Date 类型的 @db.Date 字段从 Prisma 取出后是 UTC 00:00，
  // 直接 addDays 处理即可，不会有时区误差
  return format(addDays(d, -14), 'yyyy-MM-dd')
}

// 计算距开售还有多少天（负数表示已过开售日）
export function getDaysUntilSale(departureDate: Date | string): number {
  const saleDate = getSaleDate(departureDate)
  const today = getTodayCN()
  return differenceInCalendarDays(new Date(saleDate), new Date(today))
}

// 将日期格式化为通知文本中的格式：'2026年5月1日 周五'
export function formatDateForNotify(departureDate: Date | string): string {
  const d = typeof departureDate === 'string' ? new Date(departureDate) : departureDate
  return format(d, 'yyyy年M月d日 EEEE', { locale: zhCN })
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getSaleDate, getDaysUntilSale } from '@/lib/date'

function withSaleInfo(plan: any) {
  return {
    ...plan,
    departureDate: plan.departureDate.toISOString().slice(0, 10),
    saleDate: getSaleDate(plan.departureDate),
    daysUntilSale: getDaysUntilSale(plan.departureDate),
    createdAt: plan.createdAt.toISOString(),
    updatedAt: plan.updatedAt.toISOString(),
  }
}

export async function GET() {
  const plans = await prisma.travelPlan.findMany({
    orderBy: { departureDate: 'asc' },
  })
  return NextResponse.json({ data: plans.map(withSaleInfo) })
}

export async function POST(request: NextRequest) {
  const body = await request.json()
  const plan = await prisma.travelPlan.create({
    data: {
      departureDate: new Date(body.departureDate),
      holidayName: body.holidayName || null,
      fromStation: body.fromStation,
      toStation: body.toStation,
      seatType: body.seatType,
      ticketCount: body.ticketCount ?? 1,
      status: body.status ?? 'PENDING_SALE',
      remark: body.remark || null,
    },
  })
  return NextResponse.json({ data: withSaleInfo(plan) }, { status: 201 })
}
```

**`app/api/plans/[id]/route.ts`**（编辑 + 删除）：

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { getSaleDate, getDaysUntilSale } from '@/lib/date'

type Params = { params: Promise<{ id: string }> }

function withSaleInfo(plan: any) {
  return {
    ...plan,
    departureDate: plan.departureDate.toISOString().slice(0, 10),
    saleDate: getSaleDate(plan.departureDate),
    daysUntilSale: getDaysUntilSale(plan.departureDate),
    createdAt: plan.createdAt.toISOString(),
    updatedAt: plan.updatedAt.toISOString(),
  }
}

// POST /api/plans/:id → 更新
export async function POST(request: NextRequest, { params }: Params) {
  const { id } = await params
  const body = await request.json()
  const plan = await prisma.travelPlan.update({
    where: { id },
    data: {
      ...(body.departureDate && { departureDate: new Date(body.departureDate) }),
      ...(body.holidayName !== undefined && { holidayName: body.holidayName || null }),
      ...(body.fromStation && { fromStation: body.fromStation }),
      ...(body.toStation && { toStation: body.toStation }),
      ...(body.seatType && { seatType: body.seatType }),
      ...(body.ticketCount !== undefined && { ticketCount: body.ticketCount }),
      ...(body.status && { status: body.status }),
      ...(body.remark !== undefined && { remark: body.remark || null }),
    },
  })
  return NextResponse.json({ data: withSaleInfo(plan) })
}

// DELETE /api/plans/:id → 删除
export async function DELETE(_request: NextRequest, { params }: Params) {
  const { id } = await params
  await prisma.travelPlan.delete({ where: { id } })
  return new NextResponse(null, { status: 204 })
}
```

```tsx
'use client'
import { useEffect, useState } from 'react'

  const [plans, setPlans] = useState<TravelPlan[]>([])
  const [loading, setLoading] = useState(false)
  const [modalOpen, setModalOpen] = useState(false)
  const [editingPlan, setEditingPlan] = useState<TravelPlan | null>(null)

  const fetchPlans = async () => {
    setLoading(true)

    setEditingPlan(plan)
    setModalOpen(true)
  }

    setEditingPlan(null)
    setModalOpen(true)
  }

  const handleFormSuccess = () => {
    setModalOpen(false)
    fetchPlans()
  }

        </Tag>

    },
    { title: '开售日', dataIndex: 'saleDate', width: 110 },
    {
      title: '距开售',
      dataIndex: 'daysUntilSale',

      render: v =>
```

```tsx
'use client'
import { useEffect } from 'react'
import { Modal, Form, Input, Select, DatePicker, InputNumber, message } from 'antd'
import dayjs from 'dayjs'
import type { TravelPlan } from '@/types'
import { withBasePath } from '@/lib/with-base-path'
import { SEAT_TYPE_LABELS, PLAN_STATUS_LABELS } from '@/types'

interface Props {
  open: boolean
  editingPlan: TravelPlan | null
  onSuccess: () => void
  onCancel: () => void
}

export default function PlanForm({ open, editingPlan, onSuccess, onCancel }: Props) {
  const [form] = Form.useForm()
  const isEdit = !!editingPlan

  useEffect(() => {
    if (open) {
      if (editingPlan) {
        form.setFieldsValue({
          ...editingPlan,
          departureDate: dayjs(editingPlan.departureDate),
        })
      } else {
        form.resetFields()
      }
    }
  }, [open, editingPlan, form])

  const handleOk = async () => {
    const values = await form.validateFields()
    const body = {
      ...values,
      departureDate: values.departureDate.format('YYYY-MM-DD'),
    }
    const path = isEdit ? `/api/plans/${editingPlan!.id}` : '/api/plans'
    const res = await fetch(withBasePath(path), {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    })
    if (res.ok) {
      message.success(isEdit ? '更新成功' : '新增成功')
      onSuccess()
    } else {
      message.error('操作失败，请重试')
    }
  }

  return (
    <Modal
      title={isEdit ? '编辑出行计划' : '新增出行计划'}
      open={open}
      onOk={handleOk}
      onCancel={onCancel}
      okText="保存"
      cancelText="取消"
      destroyOnHidden

        <Form.Item name="departureDate" label="出行日期" rules={[{ required: true }]}>
```

```typescript
import cron from 'node-cron'
import { prisma } from '@/lib/prisma'
import { getTodayCN, formatDateForNotify } from '@/lib/date'
import { sendWeComMessage } from '@/lib/notify'
import { addDays, parseISO } from 'date-fns'

function buildMessage(plan: {
  departureDate: Date
  holidayName: string | null
  fromStation: string
  toStation: string
}, timing: '今天' | '明天'): string {
  const dateStr = formatDateForNotify(plan.departureDate)
  const holiday = plan.holidayName ? `（${plan.holidayName}）` : ''
  return `你预定了要买${dateStr}${holiday}${plan.fromStation}到${plan.toStation}的火车票，${timing}就要开售了，请关注`
}

export function startCronJobs() {
  // 每日 00:00 China Standard Time
  // node-cron 的时区参数直接使用 IANA 时区名
  cron.schedule(
    '0 0 * * *',
    async () => {
      console.log(`[Cron] 开售提醒任务开始执行，当前时间：${new Date().toISOString()}`)

        const today = getTodayCN()

        // 今天开售：出行日期 = 今天 + 14 天
        const todaySaleDate = addDays(new Date(today), 14)
        // 明天开售：出行日期 = 今天 + 15 天
        const tomorrowSaleDate = addDays(new Date(today), 15)

        const [todayPlans, tomorrowPlans] = await Promise.all([
          prisma.travelPlan.findMany({
            where: {
              departureDate: todaySaleDate,
              status: { in: ['PENDING_SALE', 'ON_SALE'] },
            },
          }),
          prisma.travelPlan.findMany({
            where: {
              departureDate: tomorrowSaleDate,
              status: { in: ['PENDING_SALE', 'ON_SALE'] },
            },
          }),
        ])

        for (const plan of todayPlans) {
          await sendWeComMessage(buildMessage(plan, '今天'))
        }
        for (const plan of tomorrowPlans) {
          await sendWeComMessage(buildMessage(plan, '明天'))
        }

        console.log(`[Cron] 执行完成，今天开售 ${todayPlans.length} 条，明天开售 ${tomorrowPlans.length} 条`)
      } catch (err) {
        console.error('[Cron] 执行异常：', err)
      }
    },
    { timezone: 'Asia/Shanghai' }
  )

  console.log('[Cron] 定时任务已注册：每日 00:00 (Asia/Shanghai)')
}
```

**`app/api/waitlist/route.ts`**（列表，含聚合分组）：

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import type { WaitlistGroup } from '@/types'

export async function GET(request: NextRequest) {
  const sortBy = request.nextUrl.searchParams.get('sortBy') ?? 'departureTime'

  const orders = await prisma.waitlistOrder.findMany({
    orderBy: sortBy === 'price'
      ? [{ departureDate: 'asc' }, { price: 'asc' }]
      : [{ departureDate: 'asc' }, { departureTime: 'asc' }],
  })

  // 按（出发日期 + 始发站 + 终到站）聚合分组
  const groupMap = new Map<string, WaitlistGroup>()
  for (const order of orders) {
    const dateStr = order.departureDate.toISOString().slice(0, 10)
    const key = `${dateStr}|${order.fromStation}|${order.toStation}`
    if (!groupMap.has(key)) {
      groupMap.set(key, {
        groupKey: key,
        departureDate: dateStr,
        fromStation: order.fromStation,
        toStation: order.toStation,
        orders: [],
      })
    }
    groupMap.get(key)!.orders.push({
      ...order,
      departureDate: dateStr,
      price: Number(order.price),
      syncedAt: order.syncedAt.toISOString(),
      createdAt: order.createdAt.toISOString(),
    } as any)
  }

  return NextResponse.json({ data: Array.from(groupMap.values()) })
}
```

**`app/api/waitlist/sync/route.ts`**（Chrome Extension 同步数据）：

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function POST(request: NextRequest) {
  // Token 验证
  const token = request.headers.get('x-sync-token')
  if (token !== process.env.EXTENSION_SYNC_TOKEN) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const { orders } = await request.json()
  if (!Array.isArray(orders) || orders.length === 0) {
    return NextResponse.json({ error: 'orders 不能为空' }, { status: 400 })
  }

  await Promise.all(
    orders.map((order: any) =>
      prisma.waitlistOrder.upsert({
        where: { orderNo: order.orderNo },
        create: {
          orderNo: order.orderNo,
          departureDate: new Date(order.departureDate),
          fromStation: order.fromStation,
          toStation: order.toStation,
          trainNumbers: order.trainNumbers,
          seatType: order.seatType,
          ticketCount: order.ticketCount,
          price: order.price,
          departureTime: order.departureTime,
          arrivalTime: order.arrivalTime,
          status: order.status,
          rawData: order.rawData,
        },
        update: {
          trainNumbers: order.trainNumbers,
          status: order.status,
          rawData: order.rawData,
        },
      })
    )
  )

  return NextResponse.json({ success: true, count: orders.length })
}
```
